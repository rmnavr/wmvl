
(* smth ‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾\ {{{1 *)

	samplingFreq = 1024 ;	(* Hz *)
	tStart = 0 ;			(* s *)
	tEnd   = 1 ; 			(* s *)
	nPts   = (tEnd - tStart) * samplingFreq // Round;

	(* input/output signals: *)
	f0 = 1 ; (* Hz *)
	omega0 = 2 * Pi * f0 ; (* 1/s*)
	ts = Table[i/samplingFreq, {i, 0, nPts - 1}] ; (* [s] of len nPts *)
	xs = Sin[omega0*ts] + 0.0001*RandomReal[{-1, 1}, nPts];  (* [1] of len nPts*)
	ys = 1.4*Sin[omega0*ts] + 0.0001*RandomReal[{-1, 1}, nPts]; 


	(* ======================================================== *)

	ListLinePlot[{Transpose @ {ts, xs}, Transpose @ {ts, ys}}, PlotLabel -> "Signals", PlotLegends -> {"Input", "Output"}]

	(* Compute Fourier Transforms: *)
	Xs = Fourier[xs, FourierParameters -> {1, 1}] ; (* [1 + 1i] of len nPts *)
	Ys = Fourier[ys, FourierParameters -> {1, 1}] ; (* [1 + 1i] of len nPts *)
	freqs = Table[i*samplingFreq/nPts, {i, 0, nPts - 1}] ; (* [Hz] of len nPts *)

	ListLinePlot[Transpose @ {freqs, Xs}, PlotRange -> {{0, 5*f0}, Automatic}]

	(* Compute Frequency Response H(f) *)
	Hs = Ys/Xs ; (* [1 + 1i] of len nPts *)

	(* Calculate Magnitude and Phase Responses *)
	magnitudes = Abs[Hs] ;		(* [1] of len nPts*)
	phases = Arg[Hs] / Degree ; (* [°] of len nPts*)

	ListLinePlot[Transpose @ {freqs, magnitudes}, PlotRange -> {{0, 5*f0}, Automatic}]

(* DFT (numbered example) ‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾‾\ {{{1 *)

	calcDFT[xs_List] := Module[
		{nPts}
	  , nPts = Length @ xs 
	  ; (Total @ Table[xs[[n + 1]]*Exp[-I*2*Pi*#1*n/nPts], {n, 0, nPts - 1}])& /@ Table[k, {k, 0, nPts -1}]
	] ;
	(* works the same as Fourier[xs_, FourierParameters -> {1, 1}] *)
	
	xxs = {1,1,1,0,0,0,0,0} ;
	calcDFT[xxs] ;
	ListPlot[Transpose[{Range[0,7], Abs @ calcDFT @ xxs}], Filling -> Axis]
